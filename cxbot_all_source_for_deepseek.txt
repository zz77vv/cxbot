"""Audit a curated source-only release; never print sensitive literal values."""
import argparse
import ast
import hashlib
import json
from pathlib import Path
import zipfile


ROOT = Path(__file__).resolve().parents[1]


def release_files():
    top = ['main.py', 'requirements.txt', 'requirements-dev.txt', 'requirements-lock.txt',
           '.env.example', '.gitignore', 'README.md', 'pytest.ini']
    paths = [ROOT / name for name in top]
    for directory in ('telegram_bot', 'tests', 'scripts'):
        paths.extend(p for p in (ROOT / directory).rglob('*') if p.is_file()
                     and p.suffix in ('.py', '.md') and '__pycache__' not in p.parts)
    paths.extend(p for p in (ROOT / 'docs').iterdir() if p.is_file() and p.suffix in ('.md', '.json', '.xml'))
    return sorted(set(paths))


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--legacy-source', type=Path)
    parser.add_argument('--zip', type=Path)
    args = parser.parse_args()
    inventory = json.loads((ROOT / 'docs/legacy_inventory.json').read_text(encoding='utf-8'))
    secrets = []
    if args.legacy_source:
        payload = args.legacy_source.read_bytes()
        if hashlib.sha256(payload).hexdigest() != inventory['sha256']:
            raise ValueError('旧源码 SHA256 与初始检查不一致')
        tree = ast.parse(payload.decode('utf-8-sig'))
        for node in tree.body:
            if isinstance(node, ast.Assign) and any(isinstance(n, ast.Name) and n.id in {'BOT_TOKEN', 'API_HASH'} for n in node.targets):
                if isinstance(node.value, ast.Constant) and isinstance(node.value.value, str) and node.value.value:
                    secrets.append(node.value.value)
        print('Legacy SHA256: unchanged; sensitive values loaded only for in-memory matching')
    files = release_files()
    for path in files:
        text = path.read_text(encoding='utf-8-sig')
        if any(secret in text for secret in secrets):
            raise ValueError('交付文件命中源凭据；停止打包（不回显值）')
        if path.suffix == '.py':
            ast.parse(text)
    report = {
        'legacy_sha256': inventory['sha256'],
        'legacy_unchanged_checked': bool(args.legacy_source),
        'sensitive_literal_scan': 'passed' if secrets else 'no_legacy_values_supplied',
        'source_files': len(set(files) | {ROOT / 'docs/release_audit.json', ROOT / 'docs/release_manifest.json'}),
        'packaging': 'allowlist; no runtime databases/sessions/logs/env/cache/venv',
        'github': 'not configured; no repository, remote or synchronization target',
        'server': 'not accessed',
    }
    (ROOT / 'docs/release_audit.json').write_text(json.dumps(report, ensure_ascii=False, indent=2), encoding='utf-8')
    files = release_files()
    excluded = {'docs/release_manifest.json'}
    manifest = {str(p.relative_to(ROOT)).replace('\\', '/'): hashlib.sha256(p.read_bytes()).hexdigest()
                for p in files if str(p.relative_to(ROOT)).replace('\\', '/') not in excluded}
    (ROOT / 'docs/release_manifest.json').write_text(json.dumps(manifest, ensure_ascii=False, indent=2), encoding='utf-8')
    print('Release audit passed:', len(manifest), 'hashed source/test/document files')
    if args.zip:
        destination = args.zip.resolve()
        # Never overwrite an existing delivery archive.
        with zipfile.ZipFile(destination, 'x', compression=zipfile.ZIP_DEFLATED) as archive:
            for path in release_files():
                archive.write(path, 'telegram_bot_v2/' + str(path.relative_to(ROOT)).replace('\\', '/'))
        with zipfile.ZipFile(destination) as archive:
            if archive.testzip() is not None:
                raise ValueError('压缩包校验失败')
            forbidden = {'.db', '.session', '.log', '.pyc'}
            if any(Path(n).suffix in forbidden or '/.venv/' in n or n.endswith('/.env') for n in archive.namelist()):
                raise ValueError('压缩包包含运行数据，停止交付')
        print('ZIP validated:', destination.name)


if __name__ == '__main__':
    main()

    output = BytesIO()
    image.save(output, format='PNG')

        await service.close()

"""Explicitly send an operator-confirmed undelivered saved result. No userbot clients."""
import argparse
import asyncio
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))
from telegram import Bot
from telegram_bot.config import Config
from telegram_bot.logging_config import configure_logging
from telegram_bot.storage.database import Database
from telegram_bot.storage.repositories.query_repository import QueryRepository, result_from_row
from telegram_bot.storage.repositories.cache_repository import CacheRepository
from telegram_bot.services.jr.cache_service import CacheService
from telegram_bot.services.jr.export_service import ExportService
from telegram_bot.utils.process_lock import ProcessLock


async def deliver(task):
    cfg = Config.load()
    lock = ProcessLock(cfg.database)
    lock.acquire()
    handler = configure_logging(cfg)
    db = None
    try:
        db = await Database(cfg.database).open()
        repo = QueryRepository(db)
        row = await repo.get(task)
        result = result_from_row(row)
        if not row or row['state'] != 'success' or row['delivery_state'] != 'not_started' or not result:
            raise ValueError('结果不可重发；先核实并用 review_tasks 设置交付状态为 not_started')
        async with Bot(cfg.bot_token) as bot:
            exports = ExportService(cfg, None, repo, CacheService(cfg, CacheRepository(db)), bot)
            await exports.deliver(task, result)
        print('已交付保存的结果；没有重发查询或购买。')
    finally:
        if db:
            await db.close()
        handler.close()
        lock.close()


if __name__ == '__main__':
    parser = argparse.ArgumentParser()
    parser.add_argument('--task', required=True)
    args = parser.parse_args()
    asyncio.run(deliver(args.task))

"""Offline operator reconciliation. Stop the bot and verify upstream records first."""
import argparse
import asyncio
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))
from telegram_bot.storage.database import Database
from telegram_bot.storage.repositories.query_repository import QueryRepository
from telegram_bot.utils.process_lock import ProcessLock


async def review(args):
    lock = ProcessLock(args.database)
    lock.acquire()
    try:
        db = await Database(args.database).open()
        try:
            repo = QueryRepository(db)
            if args.task:
                if args.delivery:
                    row = await repo.get(args.task)
                    if not row or row['state'] != 'success':
                        raise ValueError('仅能核实已成功查询任务的交付状态')
                    await repo.delivery(args.task, args.delivery)
                else:
                    await repo.review(args.task, args.decision, args.reason or '')
                print('人工结论已记录；未执行 Telegram 请求或结果重发。')
            else:
                rows = await repo.recent_metadata()
                for row in rows:
                    print(dict(row))
        finally:
            await db.close()
    finally:
        lock.close()


if __name__ == '__main__':
    parser = argparse.ArgumentParser()
    parser.add_argument('--database', type=Path, required=True)
    parser.add_argument('--task')
    parser.add_argument('--decision', choices=('failed', 'success', 'manual'))
    parser.add_argument('--reason')
    parser.add_argument('--delivery', choices=('delivered', 'not_started'))
    args = parser.parse_args()
    if not args.database.is_file():
        parser.error('数据库不存在')
    if args.task and not (args.decision or args.delivery):
        parser.error('需要人工结论或交付结论')
    if args.task and args.decision and not args.reason:
        parser.error('需要核实原因 --reason')
    asyncio.run(review(args))

"""Read-only SQLite backup into a NEW file, suitable for migration rehearsal."""
import argparse
import sqlite3
from pathlib import Path


def snapshot(source, destination):
    source, destination = Path(source).resolve(), Path(destination).resolve()
    if not source.is_file() or source == destination:
        raise ValueError('源数据库不存在或目标与源相同')
    destination.parent.mkdir(parents=True, exist_ok=True)
    # Exclusive creation prevents accidental overwrite of the only backup.
    with destination.open('xb'):
        pass
    try:
        with sqlite3.connect(source.as_uri() + '?mode=ro', uri=True) as src:
            with sqlite3.connect(destination) as dst:
                src.backup(dst)
                if dst.execute('PRAGMA integrity_check').fetchone()[0] != 'ok':
                    raise ValueError('备份完整性检查失败')
    except BaseException:
        destination.unlink(missing_ok=True)
        raise


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--source', type=Path, required=True)
    parser.add_argument('--destination', type=Path, required=True)
    args = parser.parse_args()
    snapshot(args.source, args.destination)
    print('一致性快照已完成；源数据库未进行迁移。')


if __name__ == '__main__':
    main()

async def run(cfg):
    lock = ProcessLock(cfg.database)
    lock.acquire()
    db = Database(cfg.database)
    lifecycle = None
    log_handler = None
    try:
        log_handler = configure_logging(cfg)
        await db.open()
        manager = UserbotManager(cfg, AccountRepository(db))
        dialogue = Dialogue(cfg)
        queries = QueryRepository(db)
        cache = CacheService(cfg, CacheRepository(db))
        balances = BalanceService(manager, dialogue)
        memberships = MembershipService(MembershipRepository(db))
        selector = AccountSelector(manager, balances, memberships, queries)
        target = TargetClient(dialogue, cfg)
        application = create_application(cfg)
        exports = ExportService(cfg, dialogue, queries, cache, application.bot)
        jr = Orchestrator(manager, queries, selector, target, PaginationService(target), exports, cache)
        checkins = CheckinService(manager, CheckinRepository(db), dialogue)
        logins = LoginService(cfg, manager, lambda: checkins.start(lifecycle.registry))
        ctx = AppContext(cfg, db, manager, checkins,
                         balances, AccountStatusService(manager), jr, OCRService(cfg.ocr_timeout), cache, UserRepository(db), RequestTracker(), logins)
        lifecycle = Lifecycle(ctx, BotApplication(application, cfg), queries)
        register(application, ctx)
        loop = asyncio.get_running_loop()
        installed = []
        for sig in (signal.SIGINT, signal.SIGTERM):
            try:
                loop.add_signal_handler(sig, lifecycle.stop_event.set)
                installed.append(sig)
            except (NotImplementedError, RuntimeError):
                pass  # Windows asyncio.run handles Ctrl+C through cancellation.
        try:
            await lifecycle.run()
        finally:
            for sig in installed:
                loop.remove_signal_handler(sig)
    except Exception as error:
        logging.getLogger(__name__).error('Startup/run failed error_type=%s', type(error).__name__)
        raise
    finally:
        if lifecycle:
            await lifecycle.close()
        else:
            await db.close()
        lock.close()
        if log_handler:
            logging.getLogger().removeHandler(log_handler)
            log_handler.close()


def main():
    try:
        cfg = Config.load()
        asyncio.run(run(cfg))
    except ConfigError as error:
        print(f'配置错误：{error}', file=sys.stderr)
        raise SystemExit(2)
    except SchemaError as error:
        print(f'数据库兼容性检查失败：{error}', file=sys.stderr)
        raise SystemExit(3)
    except KeyboardInterrupt:
        pass
    except Exception as error:
        print(f'启动/运行失败：{type(error).__name__}；请查看脱敏日志。', file=sys.stderr)
        raise SystemExit(1)

import asyncio
import logging
from telegram.ext import Application
from ..handlers.registration import configure_menus

log = logging.getLogger(__name__)


def create_application(cfg):
    return (Application.builder().token(cfg.bot_token).concurrent_updates(cfg.concurrency)
            .connect_timeout(cfg.reply_timeout).read_timeout(cfg.reply_timeout)
            .write_timeout(cfg.export_timeout).build())


class BotApplication:
    def __init__(self, application, cfg):
        self.application, self.cfg = application, cfg
        self.initialize_attempted = False

    async def start(self):
        self.initialize_attempted = True
        await self.application.initialize()
        await configure_menus(self.application.bot, self.cfg)
        await self.application.start()
        await self.application.updater.start_polling(drop_pending_updates=False)

    async def stop_polling(self):
        updater = self.application.updater
        if updater and updater.running:
            await updater.stop()

    async def close(self):
        stages = [('polling', self.stop_polling)]
        if self.application.running:
            stages.append(('application-stop', self.application.stop))
        if self.initialize_attempted:
            stages.append(('application-shutdown', self.application.shutdown))
            # Also close partially initialized components if an earlier step fails.
            if self.application.updater:
                stages.append(('updater-shutdown', self.application.updater.shutdown))
            stages.append(('bot-shutdown', self.application.bot.shutdown))
        for name, close in stages:
            try:
                await asyncio.wait_for(close(), self.cfg.shutdown_timeout)
            except Exception as error:
                log.warning('Bot close stage=%s error_type=%s', name, type(error).__name__)

import asyncio
import time

def signature(message):
    return (getattr(message, 'text', '') or '',
            tuple(tuple(getattr(b, 'text', '') for b in row) for row in (getattr(message, 'buttons', None) or [])),
            str(getattr(message, 'edit_date', None)))


def find_button(message, keywords):
    for r, row in enumerate(getattr(message, 'buttons', None) or []):
        for c, button in enumerate(row):
            label = getattr(button, 'text', '') or ''
            if any(key in label for key in keywords):
                return r, c, label
    return None

    """Read-only bounded polling shared by check-in, balance and JR."""
    def __init__(self, cfg):
        self.cfg = cfg

    async def messages(self, client, bot, **kwargs):
        return await retry_safe(lambda: asyncio.wait_for(client.get_messages(bot, **kwargs), self.cfg.reply_timeout),
                                attempts=self.cfg.retry_attempts, base=self.cfg.retry_base)

    async def wait(self, client, bot, after_id, predicate, timeout, *, original=None, old_signature=None, interval=0.8):
        async def poll():
            while True:
                if original is not None:
                    refreshed = await self.messages(client, bot, ids=original.id)
                    if (refreshed and not getattr(refreshed, 'out', False)
                            and signature(refreshed) != old_signature and predicate(refreshed)):
                        return refreshed
                messages = await self.messages(client, bot, limit=10)
                for message in sorted(messages, key=lambda m: m.id):
                    if message.id > after_id and not getattr(message, 'out', False) and predicate(message):
                        return message
                await asyncio.sleep(interval)
        try:
            return await asyncio.wait_for(poll(), timeout)
        except TimeoutError:
            return None

class RequestTracker:
    def __init__(self):
        self.accepting = True
        self.active = set()

    @asynccontextmanager
    async def track(self):
        task = asyncio.current_task()
        self.active.add(task)
        try:
            yield
        finally:
            self.active.discard(task)

"""Persist only authorized keys; exclusive publication never replaces a Session."""
import os
import uuid
from pathlib import Path
from telethon.sessions import SQLiteSession


def save_authorized_session(root: Path, key: str, session) -> Path:
    if not key.isascii() or not key.isdigit():
        raise ValueError('invalid_phone')
    root.mkdir(mode=0o700, parents=True, exist_ok=True)
    pending = root / '.pending'
    pending.mkdir(mode=0o700, exist_ok=True)
    if pending.is_symlink():
        raise ValueError('unsafe_session_directory')
    pending.chmod(0o700)
    temporary = pending / (uuid.uuid4().hex + '.session')
    final = root / (key + '.session')
    fd = os.open(temporary, os.O_CREAT | os.O_EXCL | os.O_WRONLY, 0o600)
    os.close(fd)
    try:
        saved = SQLiteSession(str(temporary))
        try:
            saved.set_dc(session.dc_id, session.server_address, session.port)
            saved.auth_key = session.auth_key
            saved.save()
        finally:
            saved.close()
        temporary.chmod(0o600)
        # Same filesystem atomic no-clobber publication, unlike rename/replace.
        os.link(temporary, final)
        return final
    finally:
        for suffix in ('', '-journal', '-wal', '-shm'):
            temporary.with_name(temporary.name + suffix).unlink(missing_ok=True)

import asyncio
import logging
import time
from contextlib import asynccontextmanager
from telethon import TelegramClient
from telethon.errors import FloodWaitError
from telethon.sessions import MemorySession
from .session_store import save_authorized_session
from ..models.account import Account
from ..models.enums import AccountState
from ..utils.retry import retry_safe

log = logging.getLogger(__name__)


class AccountUnavailable(RuntimeError):
    pass


class UserbotManager:
    def __init__(self, cfg, repository, factory=None):
        self.cfg, self.repository = cfg, repository
        self.factory = factory or (lambda path: TelegramClient(str(path) if not isinstance(path, MemorySession) else path, cfg.api_id, cfg.api_hash,
                                    flood_sleep_threshold=0, request_retries=0, connection_retries=0,
                                    auto_reconnect=False, retry_delay=cfg.retry_base,
                                    raise_last_call_error=isinstance(path, MemorySession)))
        self.accounts: dict[str, Account] = {}
        self.semaphore = asyncio.Semaphore(cfg.concurrency)
        self.stopping = False

    def load(self):
        if self.accounts:
            return tuple(self.accounts.values())
        if not self.cfg.sessions.is_dir():
            log.warning('Session directory missing; no login will be requested')
            return ()
        for path in sorted(self.cfg.sessions.glob('*.session')):
            if not path.stem.isdigit() or path.is_symlink():
                log.warning('Skipping unsupported session filename')
                continue
            try:
                self.accounts[path.stem] = Account(path.stem, path, self.factory(path))
            except Exception as error:
                log.error('Session load failed error_type=%s', type(error).__name__)
                self.accounts[path.stem] = Account(path.stem, path, None, AccountState.ERROR,
                                                   error_type=type(error).__name__)
        return tuple(self.accounts.values())

    async def start(self):
        cooldowns = {r['phone']: r['until_at'] for r in await self.repository.cooldowns()}
        self.load()
        for a in self.accounts.values():
            a.unavailable_until = cooldowns.get(a.key, 0)
        await asyncio.gather(*(self.refresh(a, reconnect=True) for a in tuple(self.accounts.values())))

    def exists(self, key):
        return key in self.accounts or (self.cfg.sessions / (key + '.session')).exists()

    def login_client(self):
        return self.factory(MemorySession())

    async def enroll(self, key, session):
        if self.stopping or self.exists(key):
            raise AccountUnavailable('existing_or_stopping')
        path = save_authorized_session(self.cfg.sessions, key, session)
        account = Account(key, path, self.factory(path))
        self.accounts[key] = account
        await self.refresh(account, reconnect=True)
        if account.state != AccountState.ONLINE:
            raise AccountUnavailable('saved_but_reconnect_failed')
        return account

    async def flood_wait(self, account, seconds):
        account.unavailable_until = max(account.unavailable_until, time.time() + max(seconds, 1))
        account.state = AccountState.UNAVAILABLE
        await self.repository.set_cooldown(account.key, account.unavailable_until)

    async def refresh(self, account, *, reconnect=False):
        async with self.semaphore, account.lock:
            if account.client is None:
                account.state = AccountState.ERROR
                return account.state
            if account.unavailable_until > time.time():
                account.state = AccountState.UNAVAILABLE
                return account.state
            try:
                if not account.client.is_connected():
                    account.state = AccountState.OFFLINE
                    if not reconnect:
                        return account.state
                    await retry_safe(lambda: asyncio.wait_for(account.client.connect(), self.cfg.reply_timeout),
                                     attempts=self.cfg.retry_attempts, base=self.cfg.retry_base)
                authorized = await asyncio.wait_for(account.client.is_user_authorized(), self.cfg.reply_timeout)
                account.state = AccountState.ONLINE if authorized else AccountState.UNAUTHORIZED
                account.error_type = ''
            except asyncio.CancelledError:
                raise
            except FloodWaitError as error:
                await self.flood_wait(account, error.seconds)
            except Exception as error:
                account.state = AccountState.ERROR
                account.error_type = type(error).__name__
                log.warning('Account connection failed error_type=%s', account.error_type)
            return account.state

    @asynccontextmanager
    async def operation(self, account):
        async with self.semaphore, account.lock:
            if self.stopping or account.unavailable_until > time.time() or account.state != AccountState.ONLINE:
                raise AccountUnavailable('account_unavailable')
            if not account.client.is_connected():
                account.state = AccountState.OFFLINE
                raise AccountUnavailable('account_disconnected')
            try:
                yield account.client
            except FloodWaitError as error:
                await self.flood_wait(account, error.seconds)
                raise
            except (ConnectionError, OSError):
                account.state = AccountState.ERROR
                raise

    async def monitor(self):
        while not self.stopping:
            await asyncio.sleep(self.cfg.reconnect_interval)
            await asyncio.gather(*(self.refresh(a, reconnect=True) for a in tuple(self.accounts.values())))

    async def close(self):
        self.stopping = True
        async def disconnect(account):
            if account.client is None:
                return
            try:
                async with self.semaphore:
                    await asyncio.wait_for(account.client.disconnect(), self.cfg.reply_timeout)
            except Exception as error:
                log.warning('Disconnect failed error_type=%s', type(error).__name__)
            account.state = AccountState.OFFLINE
        await asyncio.gather(*(disconnect(a) for a in tuple(self.accounts.values())))

import os
import re
from dataclasses import dataclass, field
from pathlib import Path


class ConfigError(ValueError):
    pass


def read_private_env(path: Path) -> dict[str, str]:
    values = {}
    if not path.exists():
        return values
    for number, line in enumerate(path.read_text(encoding='utf-8-sig').splitlines(), 1):
        line = line.strip()
        if not line or line.startswith('#'):
            continue
        if '=' not in line:
            raise ConfigError(f'.env 第 {number} 行缺少等号')
        key, value = line.split('=', 1)
        if not re.fullmatch(r'[A-Z][A-Z0-9_]*', key.strip()):
            raise ConfigError(f'.env 第 {number} 行键名无效')
        values[key.strip()] = value.strip().strip('"\'')
    return values


@dataclass(frozen=True)
class Config:
    bot_token: str = field(repr=False)
    admin_id: int
    api_id: int
    api_hash: str = field(repr=False)
    database: Path
    sessions: Path
    log_file: Path
    cache: Path
    concurrency: int = 5
    jr_balance_concurrency: int = 8
    reply_timeout: float = 30
    export_timeout: float = 45
    retry_attempts: int = 3
    retry_base: float = 2
    reconnect_interval: float = 60
    cache_ttl: int = 300
    cleanup_interval: int = 60
    ocr_timeout: float = 20
    shutdown_timeout: float = 30
    login_ttl: float = 300

    @classmethod
    def load(cls, root: Path | None = None, environ=None):
        root = (root or Path(__file__).resolve().parents[1]).resolve()
        env = read_private_env(root / '.env')
        env.update(os.environ if environ is None else environ)

        def required(name):
            value = env.get(name, '').strip()
            if not value:
                raise ConfigError(f'缺少配置 {name}')
            return value

        def positive(name, default=None, cast=int):
            raw = env.get(name, default)
            try:
                value = cast(raw)
                if value <= 0 or (isinstance(value, float) and not __import__('math').isfinite(value)):
                    raise ValueError
                return value
            except (ValueError, TypeError):
                raise ConfigError(f'{name} 必须为有效正数') from None

        token = required('BOT_TOKEN')
        api_hash = required('API_HASH')
        if not re.fullmatch(r'\d+:[A-Za-z0-9_-]{20,}', token):
            raise ConfigError('BOT_TOKEN 格式错误')
        if not re.fullmatch(r'[0-9a-fA-F]{32}', api_hash):
            raise ConfigError('API_HASH 格式错误')

        def path(name, default):
            raw = env.get(name, default).strip()
            if not raw:
                raise ConfigError(f'{name} 路径为空')
            candidate = Path(raw).expanduser()
            return (root / candidate).resolve() if not candidate.is_absolute() else candidate.resolve()

        cfg = cls(token, positive('ADMIN_ID'), positive('API_ID'), api_hash,
                  path('DATABASE_PATH', 'data/bot.db'), path('SESSIONS_DIR', 'sessions'),
                  path('LOG_PATH', 'logs/bot.log'), path('CACHE_DIR', '.jr_cache_files'),
                  positive('MAX_CONCURRENCY', 5), positive('JR_BALANCE_CONCURRENCY', 8),
                  positive('REPLY_TIMEOUT', 30, float), positive('EXPORT_TIMEOUT', 45, float),
                  positive('RETRY_ATTEMPTS', 3), positive('RETRY_BASE', 2, float),
                  positive('RECONNECT_INTERVAL', 60, float),
                  ocr_timeout=positive('OCR_TIMEOUT', 20, float),
                  shutdown_timeout=positive('SHUTDOWN_TIMEOUT', 30, float),
                  login_ttl=positive('LOGIN_TTL', 300, float))
        if cfg.admin_id != 683298845:
            raise ConfigError('ADMIN_ID 必须为指定管理员 683298845')
        for protected in (cfg.database, cfg.sessions, cfg.log_file):
            if protected == cfg.cache or cfg.cache in protected.parents:
                raise ConfigError('缓存目录不得包含数据库、Session 或日志')
        if cfg.cache == root or cfg.cache in root.parents or cfg.cache == Path(cfg.cache.anchor):
            raise ConfigError('缓存目录必须为专用子目录')
        if cfg.sessions in cfg.cache.parents:
            raise ConfigError('缓存目录不得位于 Session 目录内')
        if cfg.database == cfg.log_file or cfg.sessions in cfg.database.parents or cfg.sessions in cfg.log_file.parents:
            raise ConfigError('数据库、日志和 Session 路径必须分离')
        return cfg

from functools import wraps
from telegram.ext import ApplicationHandlerStop


def authorized(user_id, admin_id):
    return user_id is not None and user_id == admin_id


def protected(ctx):
    def decorate(callback):
        @wraps(callback)
        async def wrapped(update, context):
            if not update.effective_user or not update.effective_message:
                return
            if not authorized(update.effective_user.id, ctx.config.admin_id):
                await update.effective_message.reply_text('⛔ 该功能仅限管理员使用。')
                return
            if not ctx.requests.accepting:
                await update.effective_message.reply_text('程序正在关闭，请稍后再试。')
                return
            async with ctx.requests.track():
                return await callback(update, context)
        return wrapped
    return decorate

from .guards import protected

JR_ROUTE = re.compile(r'^/jr(?:@\w+)?(?:\+|\s|$)', re.I)


def parse_request(text, bot_username=None):
    match = re.match(r'^/jr(?:@(?P<bot>\w+))?(?:(?:\+|\s+)(?P<body>.*))?$', (text or '').strip(), re.I | re.S)
    if not match:
        return None
    if match['bot'] and bot_username and match['bot'].casefold() != bot_username.casefold():
        return None
    return (match['body'] or '').strip()


def make_handler(ctx):

    return handle

class TaskRegistry:
    def __init__(self, stop_event=None):
        self.tasks = set()
        self.stop_event = stop_event
        self.failure = None

    def start(self, coroutine, name):
        task = asyncio.create_task(coroutine, name=name)
        self.tasks.add(task)
        task.add_done_callback(self._done)
        return task

    def _done(self, task):
        self.tasks.discard(task)
        if not task.cancelled() and task.exception():
            self.failure = type(task.exception()).__name__
            log.error('Background task failed name=%s error_type=%s', task.get_name(), type(task.exception()).__name__)
            if self.stop_event:
                self.stop_event.set()

    async def close(self):
        tasks = tuple(self.tasks)
        for task in tasks:
            task.cancel()
        if tasks:
            await asyncio.gather(*tasks, return_exceptions=True)
        self.tasks.clear()


class Lifecycle:
    def __init__(self, ctx, bot, queries):
        self.ctx, self.bot, self.queries = ctx, bot, queries
        self.stop_event = asyncio.Event()
        self.registry = TaskRegistry(self.stop_event)
        self.closed = False

    async def run(self):
        try:
            recovered = await self.queries.recover()
            await self.ctx.cache.repository.clear_stale_leases()
            await self.ctx.accounts.start()
            await self.bot.start()
            for row in recovered:
                await self.ctx.jr.exports.notify(row['task_id'], row['state'], 'restart_requires_review')
            await self.ctx.cache.cleanup()
            self.ctx.checkins.start(self.registry)
            self.registry.start(self.ctx.logins.loop(), 'login-expiry')
            self.registry.start(self.ctx.accounts.monitor(), 'account-monitor')
            self.registry.start(self.ctx.cache.loop(), 'cache-cleanup')
            log.info('V2 started accounts=%d', len(self.ctx.accounts.accounts))
            await self.stop_event.wait()
            if self.registry.failure:
                raise RuntimeError('critical_background_task_failed')
        finally:
            await self.close()

    async def close(self):
        if self.closed:
            return
        self.closed = True
        self.ctx.jr.accepting = False
        self.ctx.requests.accepting = False
        try:
            await self.bot.stop_polling()
        except Exception as error:
            log.warning('Stop polling error_type=%s', type(error).__name__)
        active = tuple(self.ctx.jr.active | self.ctx.requests.active)
        if active:
            _, pending = await asyncio.wait(active, timeout=self.ctx.config.shutdown_timeout)
            for task in pending:
                task.cancel()
            await asyncio.gather(*active, return_exceptions=True)
        await self.registry.close()
        for name, close in [('logins', self.ctx.logins.close), ('bot', self.bot.close), ('ocr', self.ctx.ocr.close),
                             ('accounts', self.ctx.accounts.close), ('database', self.ctx.database.close)]:
            try:
                await close()
            except Exception as error:
                log.error('Resource close failed resource=%s error_type=%s', name, type(error).__name__)

import logging
import re
from logging.handlers import RotatingFileHandler


class RedactFilter(logging.Filter):
    def __init__(self, secrets=()):
        super().__init__()
        self.secrets = tuple(s for s in secrets if s)

    def filter(self, record):
        message = record.getMessage()
        for secret in self.secrets:
            message = message.replace(secret, '[REDACTED]')
        message = re.sub(r'\b\d{5,}:[A-Za-z0-9_-]{20,}', '[REDACTED]', message)
        record.msg, record.args = message, ()
        # Exception strings may contain HTTP URLs or user query payloads.
        if record.exc_info:
            record.msg += f' error_type={record.exc_info[0].__name__}'
            record.exc_info = None
            record.exc_text = None
        return True


def configure_logging(cfg):
    cfg.log_file.parent.mkdir(parents=True, exist_ok=True)
    handler = RotatingFileHandler(cfg.log_file, maxBytes=20 * 1024 * 1024,
                                  backupCount=5, encoding='utf-8')
    handler.addFilter(RedactFilter((cfg.bot_token, cfg.api_hash)))
    handler.setFormatter(logging.Formatter('%(asctime)s %(name)s %(levelname)s %(message)s'))
    root = logging.getLogger()
    root.setLevel(logging.INFO)
    root.addHandler(handler)
    for name in ('httpx', 'httpcore', 'telethon'):
        logging.getLogger(name).setLevel(logging.WARNING)
    return handler

def parse_balance(text, target):
    match = re.search(target.balance_pattern, text or '')
    if not match:
        return OperationResult(Outcome.FORMAT_CHANGED, 'balance_pattern_not_matched')
    return OperationResult(Outcome.SUCCESS, value=int(match.group(1)))


class BalanceService:
    def __init__(self, manager, dialogue):
        self.manager, self.dialogue = manager, dialogue

    async def query(self, account, target):
        try:
            async with self.manager.operation(account) as client:
                sent = await asyncio.wait_for(client.send_message(target.bot, target.balance_cmd), self.manager.cfg.reply_timeout)
                if target.balance_button:
                    button_message = await self.dialogue.wait(client, target.bot, sent.id,
                        lambda m: bool(find_button(m, (target.balance_button,))), target.balance_timeout)
                    if not button_message:
                        return OperationResult(Outcome.FAILED, 'balance_button_missing')
                    r, c, _ = find_button(button_message, (target.balance_button,))
                    await asyncio.wait_for(button_message.click(r, c), self.manager.cfg.reply_timeout)

        except Exception as error:

            return OperationResult(Outcome.FAILED, 'balance_error', type(error).__name__)

import asyncio
import logging
import random
import time
from telethon.errors import FloodWaitError
from ..clients.dialogue import find_button, signature

class CheckinService:
    def __init__(self, manager, repository, dialogue, rng=None):
        self.manager, self.repository, self.dialogue = manager, repository, dialogue
        self.rng = rng or random.Random()
        self.tasks = {}

    async def perform(self, account, target):
        """Only causally fresh replies can confirm success. Sends/clicks never retried."""
        client = account.client
        sent = await asyncio.wait_for(client.send_message(target.bot, target.checkin_cmd), self.manager.cfg.reply_timeout)
        last_message, old_signature = None, None
        for label in target.checkin_buttons:
            message = await self.dialogue.wait(client, target.bot, sent.id,
                lambda m: bool(find_button(m, (label,))), 45,
                original=last_message, old_signature=old_signature)
            if not message:
                return OperationResult(Outcome.FAILED, 'checkin_button_missing')
            r, c, _ = find_button(message, (label,))
            old_signature = signature(message)
            last_message = message
            await asyncio.wait_for(message.click(r, c), self.manager.cfg.reply_timeout)
        # Broad legacy balance keywords alone are not proof that a button click succeeded.
        def confirmed(m):
            text = getattr(m, 'text', '') or ''
            if any(k in text for k in ('签到失败', '操作失败', '请求失败', '未获得')):
                return False
            if target.checkin_buttons and m.id == getattr(last_message, 'id', None) and signature(m) == old_signature:
                return False
            return any(k in text for k in target.success_keywords)
        reply = await self.dialogue.wait(client, target.bot,
            last_message.id if last_message else sent.id, confirmed, target.checkin_timeout,
            original=last_message, old_signature=old_signature)
        return OperationResult(Outcome.SUCCESS, 'confirmed_by_reply') if reply else OperationResult(Outcome.PENDING_REVIEW, 'checkin_reply_timeout')

    async def run_one(self, account, target):
        await self.repository.initialize(account.key, target.key, time.time(), self.rng)
        try:
            async with self.manager.operation(account):
                row = await self.repository.get(account.key, target.key)
                baseline = await self.repository.legacy_baseline(account.key, target.key)
                last = max(row['last_attempt'] or 0, row['last_success'] or 0, baseline or 0)
                now = time.time()
                if last and now < last + MIN_INTERVAL:
                    return OperationResult(Outcome.SKIPPED, 'minimum_24h_interval')
                await self.repository.attempted(account.key, target.key, now)
                try:
                    result = await self.perform(account, target)
                except asyncio.CancelledError:
                    result = OperationResult(Outcome.PENDING_REVIEW, 'cancelled_after_intent', 'CancelledError')
                    await self.repository.finish(account.key, target.key, time.time(), result, now + MIN_INTERVAL + self.rng.uniform(0, RANDOM_OFFSET_MAX))
                    raise
                except FloodWaitError as error:
                    await self.manager.flood_wait(account, error.seconds)
                    result = OperationResult(Outcome.RATE_LIMITED, 'telegram_flood_wait', 'FloodWaitError', retry_after=error.seconds)
                except TimeoutError:
                    result = OperationResult(Outcome.PENDING_REVIEW, 'checkin_transport_timeout', 'TimeoutError')
                except Exception as error:
                    log.warning('Checkin error target=%s error_type=%s', target.name, type(error).__name__)
                    result = OperationResult(Outcome.UNKNOWN, 'checkin_error', type(error).__name__)
                finished = time.time()
                next_at = max(finished + MIN_INTERVAL + self.rng.uniform(0, RANDOM_OFFSET_MAX), account.unavailable_until)
                await self.repository.finish(account.key, target.key, finished, result, next_at)
                return result
        except AccountUnavailable:
            return OperationResult(Outcome.UNAVAILABLE, 'account_unavailable')

    async def manual(self):
        async def per_account(a):
            await asyncio.sleep(self.rng.uniform(0, min(15, len(self.manager.accounts))))
            return a.key, [(t.name, await self.run_one(a, t)) for t in TARGETS]
        return await asyncio.gather(*(per_account(a) for a in self.manager.accounts.values()))

    async def scheduler(self, account, target):
        while True:
            try:
                row = await self.repository.initialize(account.key, target.key, time.time(), self.rng)
                await asyncio.sleep(max(0, row['next_at'] - time.time()))
                result = await self.run_one(account, target)
                if result.outcome in {Outcome.UNAVAILABLE, Outcome.SKIPPED}:
                    await asyncio.sleep(max(60, min(600, account.unavailable_until - time.time())))
            except asyncio.CancelledError:
                raise
            except Exception as error:
                log.error('Scheduler failure target=%s error_type=%s', target.name, type(error).__name__)
                await asyncio.sleep(60)

    def start(self, registry):
        for account in self.manager.accounts.values():
            for target in TARGETS:
                key = (account.key, target.key)
                if key not in self.tasks or self.tasks[key].done():
                    self.tasks[key] = registry.start(self.scheduler(account, target), f'checkin:{account.key}:{target.name}')

class RateLimitEncountered(RuntimeError):

class AccountSelector:
    def __init__(self, manager, balances, memberships, queries):
        self.manager, self.balances = manager, balances
        self.memberships, self.queries = memberships, queries
        self.semaphore = asyncio.Semaphore(manager.cfg.jr_balance_concurrency)

    async def rank(self):
        if any(a.unavailable_until > time.time() for a in self.manager.accounts.values()):
            raise RateLimitEncountered('telegram_flood_wait_no_account_rotation')
        async def evaluate(a):
            if a.state != AccountState.ONLINE or not await self.memberships.eligible(a.key):
                return None
            if await self.queries.unresolved_account(a.key):
                return None
            async with self.semaphore:
                balance = await self.balances.query(a, JR)
            return a, balance
        results = await asyncio.gather(*(evaluate(a) for a in self.manager.accounts.values()))
        if any(item and item[1].outcome == Outcome.RATE_LIMITED for item in results):
            raise RateLimitEncountered('telegram_flood_wait_no_account_rotation')
        return sorted((item for item in results if item and item[1].outcome == Outcome.SUCCESS),
                      key=lambda item: -item[1].value)

import asyncio
import hashlib
import logging
import re
import time
from ...models.query import JRQueryResult
from ...utils.paths import confined

log = logging.getLogger(__name__)


def normalize_jr_query(text):
    return re.sub(r'\s+', ' ', (text or '').strip()).casefold()


def jr_cache_key(text):
    return hashlib.sha256(normalize_jr_query(text).encode('utf-8')).hexdigest()


class CacheInconsistent(RuntimeError):
    pass


class CacheService:
    def __init__(self, cfg, repository):
        self.cfg, self.repository = cfg, repository
        self.lock = asyncio.Lock()

    async def get(self, key):
        # Caller holds lock until durable task references have been written.
        row = await self.repository.get(key, time.time())
        if not row:
            return None
        if row['result_type'] not in {'file', 'text'}:
            raise CacheInconsistent('invalid_cache_kind')
        if row['result_type'] == 'file':
            try:
                path = confined(self.cfg.cache, row['file_path'] or '')
                if not path.is_file():
                    raise ValueError
            except (OSError, ValueError):
                raise CacheInconsistent('cached_file_missing_or_outside_root') from None
        return JRQueryResult(row['result_type'], row['result_text'], row['file_path'], row['file_name'])

    async def cleanup(self, now=None):
        now = time.time() if now is None else now
        async with self.lock:
            active_paths = await self.repository.expire_and_references(now, self.cfg.cache_ttl)
            active = set()
            for raw in active_paths:
                try:
                    active.add(confined(self.cfg.cache, raw))
                except (ValueError, OSError):
                    log.warning('Ignored out-of-root cache reference')
            if not self.cfg.cache.is_dir():
                return 0
            deleted = 0
            for candidate in self.cfg.cache.rglob('*'):
                if candidate.is_symlink() or not candidate.is_file():
                    continue
                try:
                    path = confined(self.cfg.cache, candidate)
                    if path not in active and now - path.stat().st_mtime >= self.cfg.cache_ttl:
                        path.unlink()
                        deleted += 1
                except (OSError, ValueError):
                    log.warning('Cache cleanup skipped one unsafe/unavailable file')
            for directory in sorted(self.cfg.cache.rglob('*'), key=lambda p: len(p.parts), reverse=True):
                if directory.is_dir() and not directory.is_symlink():
                    try:
                        confined(self.cfg.cache, directory).rmdir()
                    except OSError:
                        # Only empty directories are removable; occupied directories are expected.
                        continue
            return deleted

    async def loop(self):
        while True:
            await asyncio.sleep(self.cfg.cleanup_interval)
            try:
                await self.cleanup()
            except asyncio.CancelledError:
                raise
            except Exception as error:
                log.error('Cache cleanup error_type=%s', type(error).__name__)

import asyncio

def jr_original_file_name(message):
    name = (getattr(getattr(message, 'file', None), 'name', '') or '').strip()
    name = re.split(r'[\\/]', name)[-1].strip()
    return re.sub(r'[\x00-\x1f]', '_', name) or f'jr_export_{message.id}.bin'


def jr_message_matches_export_request(message, *, baseline_id, query_after_id, target_bot_id=None):
    if getattr(message, 'out', False) or message.id <= max(baseline_id, query_after_id):
        return False
    sender = getattr(message, 'sender_id', None)
    if sender is not None and target_bot_id is not None and sender != target_bot_id:
        return False
    reply_id = getattr(message, 'reply_to_msg_id', None)
    if reply_id is None:
        reply_id = getattr(getattr(message, 'reply_to', None), 'reply_to_msg_id', None)
    if reply_id is not None and reply_id < query_after_id:
        return False
    return bool(getattr(message, 'document', None) or (getattr(message, 'file', None) and getattr(message, 'media', None)))


class ExportService:
    def __init__(self, cfg, dialogue, queries, cache, bot=None):
        self.cfg, self.dialogue, self.queries, self.cache = cfg, dialogue, queries, cache
        self.bot = bot

    async def download(self, task, client, message):
        name = jr_original_file_name(message)
        # Fixed disk basename avoids both POSIX traversal and Windows reserved device names.
        async with self.cache.lock:
            path = confined(self.cfg.cache, self.cfg.cache / uuid.uuid4().hex / 'result.bin')
            path.parent.mkdir(parents=True, exist_ok=False)
            await self.queries.attach_file(task, str(path), name)
        downloaded = await asyncio.wait_for(client.download_media(message, file=str(path)), self.cfg.export_timeout)
        actual = confined(self.cfg.cache, downloaded or path)
        if actual != path or not actual.is_file():
            raise FileNotFoundError('download_missing_or_unexpected_path')
        return JRQueryResult('file', file_path=str(path), file_name=name, export_status='member')

    async def attempt(self, task, client, message, query_after_id):
        if getattr(message, 'document', None):
            return await self.download(task, client, message), 'member', ''
        button = find_button(message, JR_EXPORT_KEYWORDS)
        if not button:
            return None, 'unknown', 'export_button_missing'
        before = await self.dialogue.messages(client, JR.bot, limit=1)
        baseline = max([message.id, *(m.id for m in before)])
        await self.queries.transition(task, QueryState.EXPORTING, source_id=message.id)
        r, c, _ = button
        answer = await asyncio.wait_for(message.click(r, c), self.cfg.reply_timeout)
        feedback = getattr(answer, 'message', '') or ''
        if any(k in feedback for k in JR_FREE_EXPORT_KEYWORDS):
            return None, 'free', 'free_export_unavailable'
        success_seen = any(k in feedback for k in JR_EXPORT_SUCCESS_KEYWORDS)

        def accepted(candidate):
            nonlocal success_seen
            text = getattr(candidate, 'text', '') or ''
            if any(k in text for k in JR_EXPORT_SUCCESS_KEYWORDS):
                success_seen = True
            return (any(k in text for k in JR_FREE_EXPORT_KEYWORDS)
                    or jr_message_matches_export_request(candidate, baseline_id=baseline, query_after_id=query_after_id,
                                                         target_bot_id=getattr(message, 'sender_id', None)))
        candidate = await self.dialogue.wait(client, JR.bot, baseline, accepted, self.cfg.export_timeout)
        if candidate and any(k in (getattr(candidate, 'text', '') or '') for k in JR_FREE_EXPORT_KEYWORDS):
            return None, 'free', 'free_export_unavailable'
        if candidate:
            try:
                return await self.download(task, client, candidate), 'member', ''
            except (OSError, TimeoutError) as error:
                log.warning('Export download failed task=%s error_type=%s', task, type(error).__name__)
                return None, 'member', 'export_download_failed'
        return None, 'member' if success_seen else 'unknown', 'export_file_timeout'

"""Offline OCR compatibility. No Telegram clicks or automatic verification submission."""
import asyncio
import logging
import re
import unicodedata
from concurrent.futures import ThreadPoolExecutor
from dataclasses import dataclass
from io import BytesIO
from PIL import Image, ImageOps
import warnings
# This dependency changes the global warning filters during import.
with warnings.catch_warnings():
    import ddddocr
from ...target_bots import OCR_MIN_CONFIDENCE, OCR_MAX_ATTEMPTS, OCR_IGNORE_BUTTON_KEYWORDS
from ...models.enums import Outcome

log = logging.getLogger(__name__)

def normalize_ocr_text(text: str) -> str:
    """统一全半角并去掉 OCR/按钮中的空格、符号和表情。"""
    normalized = unicodedata.normalize("NFKC", text or "")
    return "".join(
        ch for ch in normalized
        if ch.isalnum() or "\u4e00" <= ch <= "\u9fff"
    ).lower()

def extract_ocr_candidates(message):
    """提取验证码候选按钮，排除刷新、分页、购买等功能按钮。"""
    candidates = []
    for row_idx, row in enumerate(getattr(message, "buttons", None) or []):
        for col_idx, button in enumerate(row):
            label = (getattr(button, "text", None) or "").strip()
            normalized = normalize_ocr_text(label)
            if not normalized:
                continue
            if any(keyword in label for keyword in OCR_IGNORE_BUTTON_KEYWORDS):
                continue
            candidates.append((row_idx, col_idx, label, normalized))
    return candidates

def _pil_to_png_bytes(image: Image.Image) -> bytes:
    output = BytesIO()
    image.save(output, format="PNG")
    return output.getvalue()

def _classification_with_confidence(image_bytes: bytes, ocr_engine):
    """执行一次 OCR，并从概率结果中计算平均置信度。"""
    result = ocr_engine.classification(
        image_bytes,
        probability=True,
        png_fix=True,
    )

    if isinstance(result, str):
        return result, None

    # ddddocr 1.6+ 直接返回已解码文本和总置信度。
    if "text" in result:
        confidence = result.get("confidence")
        return (
            str(result.get("text") or ""),
            float(confidence) if confidence is not None else None,
        )

    # 兼容旧版 ddddocr 的概率结果结构。
    charsets = result.get("charsets") or []
    probabilities = result.get("probability") or []
    recognized = []
    scores = []
    for row in probabilities:
        if not row:
            continue
        index = max(range(len(row)), key=row.__getitem__)
        if index >= len(charsets):
            continue
        char = charsets[index]
        if char:
            recognized.append(char)
            scores.append(float(row[index]))

    confidence = sum(scores) / len(scores) if scores else 0.0
    return "".join(recognized), confidence

def _recognize_captcha_sync(
    image_bytes: bytes,
    candidate_charset: str,
    candidate_count: int,
    ocr_engine,
):
    """
    识别“请点击第 N 个字”类型验证码。

    先从上半部分的红色数字识别 N，再按照图片中的等宽位置裁出第 N
    个汉字。Telegram 按钮的排列顺序完全不参与位置判断。
    """
    if ocr_engine is None:
        ocr_engine = ddddocr.DdddOcr(beta=True, show_ad=False)

    with Image.open(BytesIO(image_bytes)) as source:
        image = source.convert("RGB")

    width, height = image.size
    if width < 100 or height < 80:
        raise RuntimeError(f"验证码图片尺寸异常：{width}x{height}")

    # 顶部要求中的序号是红色。只保留红色像素，可以避免正文汉字和
    # 下方 1/2/3/4 编号干扰序号识别。
    instruction = image.crop((0, 0, width, max(1, int(height * 0.48))))
    red_mask = Image.new("L", instruction.size, 255)
    red_mask.putdata([
        0 if (
            red >= 110
            and red - green >= 35
            and red - blue >= 35
            and red >= green * 1.25
            and red >= blue * 1.25
        ) else 255
        for red, green, blue in instruction.getdata()
    ])

    red_bbox = ImageOps.invert(red_mask).getbbox()
    if red_bbox:
        left, top, right, bottom = red_bbox
        margin_x = max(4, int((right - left) * 0.15))
        margin_y = max(4, int((bottom - top) * 0.15))
        red_mask = red_mask.crop((
            max(0, left - margin_x),
            max(0, top - margin_y),
            min(red_mask.width, right + margin_x),
            min(red_mask.height, bottom + margin_y),
        ))
    else:
        # 极端情况下颜色压缩严重，退回到整个顶部区域识别数字。
        red_mask = ImageOps.autocontrast(ImageOps.grayscale(instruction))

    ocr_engine.set_ranges("123456789")
    index_text = ocr_engine.classification(_pil_to_png_bytes(red_mask))
    index_digits = re.findall(r"[1-9]", str(index_text))
    valid_indexes = [
        int(value) for value in index_digits
        if 1 <= int(value) <= candidate_count
    ]
    if len(set(valid_indexes)) != 1:
        raise RuntimeError(
            f"OCR 无法确定图片要求的第几个字：{index_text!r}"
        )
    target_index = valid_indexes[0]

    # 下半部分按候选数量等分。每格左侧约 35% 是“1.”这样的编号，
    # 右侧才是目标字；裁掉编号后只识别图片顺序中的第 N 个汉字。
    slot_width = width / candidate_count
    slot_left = (target_index - 1) * slot_width
    word_crop = image.crop((
        int(slot_left + slot_width * 0.35),
        int(height * 0.45),
        int(slot_left + slot_width * 0.95),
        int(height * 0.96),
    ))
    word_crop = ImageOps.grayscale(word_crop)
    word_crop = ImageOps.autocontrast(word_crop)
    word_crop = word_crop.point(lambda value: 0 if value < 125 else 255)

    ocr_engine.set_ranges(candidate_charset)
    recognized, confidence = _classification_with_confidence(
        _pil_to_png_bytes(word_crop), ocr_engine
    )
    return target_index, recognized, confidence


@dataclass(frozen=True)
class OCRResult:
    outcome: Outcome
    text: str = ''
    confidence: float | None = None
    target_index: int | None = None
    error_type: str = ''


class OCRService:
    def __init__(self, timeout=20, engine_factory=None):
        self.timeout = timeout
        self.engine_factory = engine_factory or (lambda: ddddocr.DdddOcr(beta=True, show_ad=False))
        self.engine = None
        self.executor = ThreadPoolExecutor(max_workers=1, thread_name_prefix='jr-ocr')
        self.lock = asyncio.Lock()
        self.pending = None
        self.closed = False
        self.range_restricted = False

    def _work(self, image_bytes, charset, count, positional):
        if self.engine is None or (not positional and not charset and self.range_restricted):
            self.engine = self.engine_factory()
            self.range_restricted = False
        if positional:
            self.range_restricted = True
            return _recognize_captcha_sync(image_bytes, charset, count, self.engine)
        with Image.open(BytesIO(image_bytes)) as source:
            payload = _pil_to_png_bytes(source.convert('RGB'))
        if charset:
            self.engine.set_ranges(charset)
            self.range_restricted = True
        text, confidence = _classification_with_confidence(payload, self.engine)
        return None, text, confidence

    async def recognize(self, image_bytes, charset='', count=0, *, positional=False):
        async with self.lock:
            if self.closed or (self.pending is not None and not self.pending.done()):
                return OCRResult(Outcome.UNAVAILABLE, error_type='OCRBusyOrClosed')
            if positional and (count < 1 or not charset):
                return OCRResult(Outcome.FAILED, error_type='InvalidCandidates')
            self.pending = asyncio.get_running_loop().run_in_executor(
                self.executor, self._work, image_bytes, charset, count, positional)
            # Observe late failures after timeout without losing the busy flag.
            self.pending.add_done_callback(lambda future: future.exception() if not future.cancelled() else None)
            try:
                index, text, confidence = await asyncio.wait_for(asyncio.shield(self.pending), self.timeout)
                if not normalize_ocr_text(text):
                    return OCRResult(Outcome.FAILED, error_type='EmptyRecognition')
                if confidence is not None and confidence < OCR_MIN_CONFIDENCE:
                    return OCRResult(Outcome.FAILED, text, confidence, index, 'LowConfidence')
                return OCRResult(Outcome.SUCCESS, text, confidence, index)
            except TimeoutError:
                log.warning('OCR recognition timeout')
                return OCRResult(Outcome.TIMEOUT, error_type='TimeoutError')
            except Exception as error:
                log.warning('OCR recognition error_type=%s', type(error).__name__)
                return OCRResult(Outcome.FAILED, error_type=type(error).__name__)

    async def recognize_captcha(self, image_bytes, candidate_charset, candidate_count):
        """Legacy tuple adapter for offline images; never resolves remote security checks."""
        result = await self.recognize(image_bytes, candidate_charset, candidate_count, positional=True)
        if result.outcome != Outcome.SUCCESS:
            raise RuntimeError(result.error_type)
        return result.target_index, result.text, result.confidence

    async def close(self):
        self.closed = True
        if self.pending and not self.pending.done():
            try:
                await asyncio.wait_for(asyncio.shield(self.pending), self.timeout)
            except TimeoutError:
                log.error('OCR native worker still running at shutdown')
            except Exception as error:
                log.warning('OCR worker shutdown error_type=%s', type(error).__name__)
        self.executor.shutdown(wait=False, cancel_futures=True)

import asyncio
import logging
import time
from telethon.errors import FloodWaitError
from ...clients.userbot_manager import AccountUnavailable
from ...models.enums import QueryState
from ...models.query import JRQueryResult, QueryReport

class Orchestrator:
    def __init__(self, manager, repository, selector, target, pagination, exports, cache):
        self.manager, self.repository, self.selector = manager, repository, selector
        self.target, self.pagination, self.exports, self.cache = target, pagination, exports, cache
        self.locks = KeyedLocks()
        self.accepting = True
        self.active = set()

    async def execute(self, request_id, request_text):
        if not self.accepting:
            return QueryReport('', QueryState.FAILED, 'shutting_down')
        if not request_text.strip():
            return QueryReport('', QueryState.FAILED, 'empty_query')
        current_task = asyncio.current_task()
        self.active.add(current_task)
        try:
            async with self.locks.hold(jr_cache_key(request_text)):
                row, fresh = await self.repository.claim(str(request_id), jr_cache_key(request_text), time.time())
                task = row['task_id']
                if not fresh:
                    return QueryReport(task, QueryState(row['state']), 'existing_task_no_resubmission', duplicate=True)
                return await self._run(task, request_text)
        finally:
            self.active.discard(current_task)

    async def _run(self, task, text):
        try:
            async with self.cache.lock:
                cached = await self.cache.get(jr_cache_key(text))
                if cached:
                    await self.repository.complete(task, cached, time.time(), self.manager.cfg.cache_ttl, from_cache=True)
            if cached:
                return await self._deliver(task, cached, 'cache_hit')
            ranked = await self.selector.rank()
            if not ranked:
                await self.repository.transition(task, QueryState.FAILED, 'no_available_balance')
                return QueryReport(task, QueryState.FAILED, 'no_available_balance')
            # Only one chosen account may receive the request. A limit never triggers rotation.
            account = ranked[0][0]
            async with self.manager.operation(account) as client:
                if not await self.repository.reserve(task, account.key, time.time()):
                    await self.repository.transition(task, QueryState.FAILED, 'account_busy_or_daily_limit')
                    return QueryReport(task, QueryState.FAILED, 'account_busy_or_daily_limit')
                sent = await self.target.submit(client, text)
                await self.repository.transition(task, QueryState.SUBMITTED, sent_id=sent.id)
                reply = await self.target.ready(client, sent.id)
                if not reply:
                    return await self._hold(task, QueryState.PENDING_REVIEW, 'query_reply_timeout')
                await self.repository.transition(task, QueryState.SUBMITTED, source_id=reply.id)
                kind = classify(reply)
                if kind == 'manual':
                    return await self._hold(task, QueryState.MANUAL, 'security_verification_requires_human')
                if kind == 'failed':
                    await self.repository.transition(task, QueryState.FAILED, 'explicit_target_failure')
                    return QueryReport(task, QueryState.FAILED, 'explicit_target_failure')
                if kind != 'result':
                    return await self._hold(task, QueryState.PENDING_REVIEW, 'unrecognized_target_reply')
                buy = find_buy_button(reply)
                if buy:
                    await self.repository.transition(task, QueryState.PURCHASING, source_id=reply.id)
                    reply = await self.target.click_and_wait(client, reply, buy)
                    if not reply:
                        return await self._hold(task, QueryState.PENDING_REVIEW, 'purchase_reply_timeout')
                    if classify(reply) == 'manual':
                        return await self._hold(task, QueryState.MANUAL, 'verification_after_purchase')
                    if classify(reply) != 'result' or find_buy_button(reply):
                        return await self._hold(task, QueryState.PENDING_REVIEW, 'purchase_not_confirmed')
                exported, status, reason = await self.exports.attempt(task, client, reply, sent.id)
                if exported:
                    result = exported
                else:
                    await self.repository.transition(task, QueryState.COLLECTING, source_id=reply.id)
                    content = await self.pagination.collect(client, reply)
                    if not content:
                        return await self._hold(task, QueryState.PENDING_REVIEW, 'empty_result')
                    result = JRQueryResult('text', content, export_status=status, fallback_reason=reason)
                async with self.cache.lock:
                    await self.repository.complete(task, result, time.time(), self.manager.cfg.cache_ttl)
            return await self._deliver(task, result, 'query_completed')
        except asyncio.CancelledError:
            row = await self.repository.get(task)
            if row['state'] != 'success':
                state = QueryState.FAILED if row['state'] == 'prepared' else QueryState.PENDING_REVIEW
                await self.repository.transition(task, state, 'cancelled_before_submission' if state == QueryState.FAILED else 'cancelled_after_intent')
            raise
        except (CacheInconsistent, ManualVerification) as error:
            return await self._hold(task, QueryState.MANUAL, str(error))
        except RateLimitEncountered:
            await self.repository.transition(task, QueryState.FAILED, 'telegram_flood_wait_no_submission')
            return QueryReport(task, QueryState.FAILED, 'telegram_flood_wait_no_submission')
        except AccountUnavailable:
            await self.repository.transition(task, QueryState.FAILED, 'account_unavailable_before_submission')
            return QueryReport(task, QueryState.FAILED, 'account_unavailable_before_submission')
        except Exception as error:
            log.error('JR error task=%s error_type=%s', task, type(error).__name__)
            row = await self.repository.get(task)
            if row['state'] == 'success':
                return QueryReport(task, QueryState.SUCCESS, 'delivery_requires_review', result_from_row(row))
            if row['state'] == 'prepared':
                await self.repository.transition(task, QueryState.FAILED, 'error_before_submission')
                return QueryReport(task, QueryState.FAILED, 'error_before_submission')
            return await self._hold(task, QueryState.PENDING_REVIEW, f'uncertain_{type(error).__name__}')

    async def _hold(self, task, state, reason):
        await self.repository.transition(task, state, reason)
        await self.exports.notify(task, state.value, reason)
        return QueryReport(task, state, reason)

    async def _deliver(self, task, result, reason):
        try:
            await self.exports.deliver(task, result)
        except asyncio.CancelledError:
            raise
        except Exception as error:
            log.error('Delivery uncertain task=%s error_type=%s', task, type(error).__name__)
            await self.exports.notify(task, 'delivery_unknown', 'result_saved_no_automatic_resend')
            return QueryReport(task, QueryState.SUCCESS, 'delivery_requires_review', result)
        return QueryReport(task, QueryState.SUCCESS, reason, result)

import asyncio
from .result_parser import (parse_page_info, clean_page_text, page_content_signature,
                            find_next_page_button, is_verification_message, classify)
from ...target_bots import JR_MAX_PAGES, JR_PAGE_INTERVAL


class PaginationService:
    def __init__(self, target):
        self.target = target

    async def collect(self, client, first_message, max_pages=JR_MAX_PAGES):
        if not 1 <= max_pages <= JR_MAX_PAGES:
            raise ValueError('分页上限不得扩大')
        pages, seen, visited = {}, set(), set()
        current, expected_total = first_message, None
        for index in range(max_pages):
            if is_verification_message(current):
                raise ManualVerification('verification_during_pagination')
            if classify(current) != 'result' and index == 0:
                raise ValueError('unrecognized_result')
            info = parse_page_info(current)
            content = page_content_signature(current)
            if content in seen:
                raise ValueError('duplicate_page_content')
            seen.add(content)
            if info:
                number, total = info
                if number in visited or (pages and number != max(pages) + 1) or (not pages and number != 1):
                    raise ValueError('duplicate_or_missing_page')
                if expected_total is not None and expected_total != total:
                    raise ValueError('page_total_changed')
                expected_total = total
                visited.add(number)
            else:
                if expected_total is not None:
                    raise ValueError('page_marker_disappeared')
                number, total = index + 1, None
            pages[number] = clean_page_text(current)
            if info and number == total:
                return '\n\n'.join(pages.values()).strip()
            button = find_next_page_button(current)
            if not button:
                if info and number < total:
                    raise ValueError('next_page_button_missing')
                return '\n\n'.join(pages.values()).strip()
            if index + 1 == max_pages:
                raise ValueError('page_limit_exceeded')
            changed = await self.target.click_and_wait(client, current, button)
            if not changed:
                raise TimeoutError('next_page_timeout')
            if (getattr(changed, 'text', '') or '') == (getattr(current, 'text', '') or ''):
                raise ValueError('page_did_not_advance')
            new_info = parse_page_info(changed)
            if info and new_info and new_info[0] <= info[0]:
                raise ValueError('page_did_not_advance')
            current = changed
            await asyncio.sleep(JR_PAGE_INTERVAL)
        raise ValueError('page_limit_exceeded')


class ManualVerification(RuntimeError):
    pass
